export default { async fetch(request) { const url = new URL(request.url); // 处理根路径 / ，确认 Worker 是否运行 if (url.pathname === '/' || url.pathname === '') { return new Response('Worker is running. Use: /suggest?engine=google&q=你的查询', { status: 200, headers: { 'Content-Type': 'text/plain;charset=UTF-8' } }); } // 只处理 /suggest 路径（可选：如果你想让 /suggest 以外也 fallback，可以去掉这个 if） if (url.pathname !== '/suggest') { return new Response('Use /suggest path. Example: /suggest?engine=google&q=test', { status: 404, headers: { 'Content-Type': 'text/plain;charset=UTF-8' } }); } const engine = url.searchParams.get('engine') || 'google'; const query = url.searchParams.get('q') || ''; if (!query) { return new Response(JSON.stringify({ error: 'Missing q parameter' }), { status: 400, headers: { 'Content-Type': 'application/json;charset=UTF-8' } }); } let targetUrl; const headers = { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36' }; switch (engine.toLowerCase()) { case 'google': targetUrl = `https://suggestqueries.google.com/complete/search?client=firefox&hl=zh-CN&q=${encodeURIComponent(query)}`; break; case 'baidu': targetUrl = `https://suggestion.baidu.com/su?wd=${encodeURIComponent(query)}&cb=callback`; break; case 'bing': targetUrl = `https://api.bing.com/qsonhs.aspx?type=cb&q=${encodeURIComponent(query)}`; break; case 'duckduckgo': targetUrl = `https://ac.duckduckgo.com/ac/?q=${encodeURIComponent(query)}`; break; default: return new Response(JSON.stringify({ error: 'Unsupported engine' }), { status: 400, headers: { 'Content-Type': 'application/json;charset=UTF-8' } }); } const corsHeaders = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, OPTIONS', 'Access-Control-Max-Age': '86400', }; if (request.method === 'OPTIONS') { return new Response(null, { headers: corsHeaders }); } try { const response = await fetch(targetUrl, { headers }); if (!response.ok) { throw new Error(`Upstream ${response.status}`); } let data = await response.text(); let suggestions = []; if (engine === 'google') { const json = JSON.parse(data); suggestions = json[1] || []; } else if (engine === 'baidu') { data = data.replace(/^callback\(/, '').replace(/\);?$/, ''); const json = JSON.parse(data); suggestions = json.s || []; } else if (engine === 'bing') { data = data.replace(/^QSONHS\.callback\(/, '').replace(/\);?$/, ''); const json = JSON.parse(data); suggestions = (json[0] || []).map(item => item[0]); } else if (engine === 'duckduckgo') { const json = JSON.parse(data); suggestions = json.map(item => item.phrase); } const unified = { suggestions }; return new Response(JSON.stringify(unified), { headers: { ...corsHeaders, 'Content-Type': 'application/json;charset=UTF-8', }, }); } catch (err) { return new Response(JSON.stringify({ error: `Proxy error: ${err.message}` }), { status: 500, headers: { ...corsHeaders, 'Content-Type': 'application/json;charset=UTF-8', }, }); } } };
